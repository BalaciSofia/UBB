// c_tcp_server.c
#include <stdio.h>
#include <string.h>
#include <stdint.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <netinet/in.h>
#include <sys/socket.h>

int main() {
    char ip[32];
    int port;

    printf("IP to bind: ");
    scanf("%31s", ip);
    printf("Port: ");
    scanf("%d", &port);

    int s = socket(AF_INET, SOCK_STREAM, 0);

    struct sockaddr_in srv;
    memset(&srv, 0, sizeof(srv));
    srv.sin_family = AF_INET;
    srv.sin_port = htons(port);
    inet_pton(AF_INET, ip, &srv.sin_addr);

    bind(s, (struct sockaddr*)&srv, sizeof(srv));
    listen(s, 5);
    printf("TCP server listening on %s %d\n", ip, port);

    while (1) {
        int c = accept(s, NULL, NULL);
        if (c < 0) continue;

        // receiving string
        int32_t ln_net;
        recv(c, &ln_net, 4, 0);
        int32_t ln = ntohl(ln_net);

        char msg[1024];
        if (ln > 1023) ln = 1023;
        recv(c, msg, ln, 0);
        msg[ln] = 0;
        printf("string: %s\n", msg);

        // receiving int
        /*
        int32_t n_net;
        recv(c, &n_net, 4, 0);
        int32_t n = ntohl(n_net);
        printf("int: %d\n", n);
        */

        // receiving float (32-bit)
        /*
        uint32_t f_net;
        recv(c, &f_net, 4, 0);
        f_net = ntohl(f_net);
        float f;
        memcpy(&f, &f_net, 4);
        printf("float: %f\n", f);
        */

        // receiving char
        /*
        char ch;
        recv(c, &ch, 1, 0);
        printf("char: %c\n", ch);
        */

        // receiving array of ints
        /*
        int32_t count_net;
        recv(c, &count_net, 4, 0);
        int32_t count = ntohl(count_net);
        int32_t arr[256];
        if (count > 256) count = 256;
        for (int i = 0; i < count; i++) {
            int32_t x_net;
            recv(c, &x_net, 4, 0);
            arr[i] = ntohl(x_net);
        }
        printf("array: ");
        for (int i = 0; i < count; i++) printf("%d ", arr[i]);
        printf("\n");
        */

        // sending string back
        char reply[1060];
        snprintf(reply, sizeof(reply), "OK: %s", msg);
        int32_t rlen = strlen(reply);
        int32_t rlen_net = htonl(rlen);
        send(c, &rlen_net, 4, 0);
        send(c, reply, rlen, 0);

        close(c);
    }

    close(s);
    return 0;
}
