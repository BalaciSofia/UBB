1.
customer has one account 

an account has multiple food types and many orders(order /refund)

statistics include the number of order/refund, total ops, money left(initial deposit- spent)

create table food(
	foodID int primary key,
	foodName varchar(50)
)

create table account(
	accountID int primary key,
	deposit decimal(10,2)
)

create table customer(
	customerID int primary key,
	accountID int foreign key rferences account(accountID)
)


create table order(
	orderID int primary key,
	status varchar(10), --order/refund
	accountID int,
	constraint foreign key (accountID) references account(accountID)
)


create table accountFood (
	accountID int,
	foodID int,
	primary key(accountID,foodID),
	constraint foreign key (accountID) references account(accountID),
	constraint foreign key (foodID) references food(foodID)
)


creat table statistics(
	statId int primary key,
	noOrders int,
	noRefunds int,
	totalOp as (noOrders + noRefunds) persisted,
	modey decimal(10,2),
	accountId int,
	constraint foreign key (accountID) references account(accountID)
)

2.
create proc problem2 (@customerid int)
as
begin
	--all food names ordered by customer and the amount of money left
	Select f.foodName,s.money
	from costumer c 
		inner join account a on c.accountID=a.accountID
		inner join accountFood af on a.accountId=af.accountID
		inner join food f on af.foodID=f.foodID 
		inner join statistics s on a.customerID=s.accountID
	where c.customerID=@customerID
go
