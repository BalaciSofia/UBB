-- 1) Habitats (no FK dependencies)

CREATE TABLE dbo.Habitats (
    HabitatID INT PRIMARY KEY,
    Name VARCHAR(100) ,
    Climate VARCHAR(50) ,        
    AreaSize DECIMAL(10,2) 
);

-- 2) Species (each species linked to one habitat)

CREATE TABLE Species (
    SpeciesID INT PRIMARY KEY,
    Name VARCHAR(100) ,
    Classification VARCHAR(50),
    DietType VARCHAR(50),
    HabitatID INT,
    CONSTRAINT FK_Species_Habitat FOREIGN KEY (HabitatID) REFERENCES Habitats(HabitatID)
);

-- 3) Animals (each animal belongs to one species)
CREATE TABLE Animals (
    AnimalID INT  PRIMARY KEY,
    Name VARCHAR(100),
    Age INT,
    Gender VARCHAR(10),
    SpeciesID INT,
    CONSTRAINT FK_Animal_Species FOREIGN KEY (SpeciesID) REFERENCES Species(SpeciesID)
);

-- 4) Staff
CREATE TABLE Staff (
    StaffID INT PRIMARY KEY,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    Role VARCHAR(50),  
    HireDate DATE 
);

-- 5) StaffHabitat (many-to-many: staff <-> habitats)
CREATE TABLE StaffHabitat (
    StaffID INT,
    HabitatID INT,
    AssignedDate DATE,
    PRIMARY KEY (StaffID, HabitatID),
    CONSTRAINT FK_StaffHabitat_Staff FOREIGN KEY (StaffID) REFERENCES Staff(StaffID),
    CONSTRAINT FK_StaffHabitat_Habitat FOREIGN KEY (HabitatID) REFERENCES Habitats(HabitatID)
);

-- 6) Food (independent)
CREATE TABLE Food (
    FoodID INT PRIMARY KEY,
    Name VARCHAR(100),
    Type VARCHAR(50),
    QuantityInStock INT 
);


-- 7) Visitors (independent)
CREATE TABLE Visitors (
    VisitorID INT PRIMARY KEY,
    FirstName VARCHAR(50),
    LastName VARCHAR(50),
    Age INT
);

-- 8) Donations (visitor -> donations) (one visitor can make many donations)
CREATE TABLE Donations (
    DonationID INT PRIMARY KEY,
    VisitorID INT,
    Amount DECIMAL(10,2),
    DonationDate DATETIME,
    Purpose VARCHAR(150),
    CONSTRAINT FK_Donations_Visitor FOREIGN KEY (VisitorID) REFERENCES Visitors(VisitorID)
);

-- 9) Shop (independent)
CREATE TABLE Shop (
    ProductID INT PRIMARY KEY,
    ProductName VARCHAR(150),
    Price DECIMAL(10,2),
    Stock INT 
);

-- 10) Events (linked to Habitat)
CREATE TABLE Events (
    EventID INT PRIMARY KEY,
    Name VARCHAR(150),
    Description VARCHAR(300),
    EventDate DATETIME,
    HabitatID INT,
    CONSTRAINT FK_Events_Habitat FOREIGN KEY (HabitatID) REFERENCES Habitats(HabitatID)
);
